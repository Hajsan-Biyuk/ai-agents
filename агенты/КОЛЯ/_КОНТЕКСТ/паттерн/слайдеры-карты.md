## Яндекс.Карта (не map.png)

### HTML

```html
<div class="contacts__map">
  <div
    id="map"
    data-lat="45.30861"
    data-lng="34.64750"
    data-zoom="14"
    data-pin="img/icon-map-pin.svg"
  ></div>
</div>
```

### Подключение

```html
<script src="https://api-maps.yandex.ru/2.1/?lang=ru_RU"></script>
<script src="js/common.js?ver=1.0"></script>
```

### JS

```javascript
if ($('#map').length && typeof ymaps !== 'undefined') {
  ymaps.ready(initYandexMap);
}

function initYandexMap() {
  var $mapEl = $('#map');
  var lat = parseFloat($mapEl.data('lat'), 10) || 45.30861;
  var lng = parseFloat($mapEl.data('lng'), 10) || 34.64750;
  var zoom = parseInt($mapEl.data('zoom'), 10) || 14;
  var pinHref = $mapEl.data('pin') || 'img/icon-map-pin.svg';

  var myMap = new ymaps.Map('map', {
    center: [lat, lng],
    zoom: zoom
  });

  var placemark = new ymaps.Placemark([lat, lng], {}, {
    iconLayout: 'default#image',
    iconImageHref: pinHref,
    iconImageSize: [24, 32],
    iconImageOffset: [-12, -32]
  });

  myMap.controls.remove('searchControl');
  myMap.behaviors.disable('scrollZoom');
  myMap.geoObjects.add(placemark);
}
```

## Плавный скролл к якорям

```javascript
function getAnchorOffset() {
  return $('.header').outerHeight() || 0;
}

$('a[href^="#"]').not('.fancybox, .tabs-page a').on('click', function (e) {
  var id = $(this).attr('href');
  if (id.length < 2) return;
  var $target = $(id);
  if (!$target.length) return;
  e.preventDefault();
  $('html, body').animate({
    scrollTop: $target.offset().top - getAnchorOffset()
  }, 500);
});
```
